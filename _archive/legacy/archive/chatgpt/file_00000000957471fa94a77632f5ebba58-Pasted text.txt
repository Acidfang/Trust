def escape_xml(text: str) -> str:
    """Escape XML special characters - UNIVERSAL for all output formats."""
    if text is None:
        return ""
    replacements = {
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&apos;"
    }
    result = str(text)
    for char, escaped in replacements.items():
        result = result.replace(char, escaped)
    return result

# ========== TRANSLATION LAYER (Song to User Format) ==========

def translate_song_to_format(song: dict, target_format: str) -> Any:
    """
    LAYER 2: Translate canonical song to user-expected output format.
    
    Supported formats:
    - "song": Return raw canonical song (internal)
    - "svg": SVG visualization (default)
    - "json": JSON representation
    - "markdown": Human-readable markdown
    - "text": Plain text summary
    - "symbol": Symbols only (universal recovery format)
    - "verse": Verse only (human memorable)
    
    Args:
        song: Canonical song structure from generate_render_song()
        target_format: Output format
    
    Returns:
        Formatted output appropriate to user context
    """
    if target_format == "song":
        return song
    
    elif target_format == "verse":
        return song["canonical"]["verse"]
    
    elif target_format == "symbol":
        return song["canonical"]["symbols"]
    
    elif target_format == "json":
        return {
            "principle": song["metadata"]["principle"],
            "operation": song["metadata"]["operation"],
            "type": song["metadata"]["container_type"],
            "verse": song["canonical"]["verse"],
            "symbols": song["canonical"]["symbols"],
            "timestamp": song["metadata"]["timestamp"]
        }
    
    elif target_format == "markdown":
        return f"""## {song["metadata"]["container_type"].upper()}

**Principle**: {song["metadata"]["principle"]}

### Verse
{song["canonical"]["verse"]}

### Symbols
```
{song["canonical"]["symbols"]}
```

**Operation**: {song["metadata"]["operation"]}  
**Generated**: {song["metadata"]["timestamp"]}
"""
    
    elif target_format == "text":
        return f"[{song['metadata']['principle']}] {song['canonical']['verse']}"
    
    else:  # Default: "svg" or unknown → return SVG
        return None  # Will be handled by existing render functions

# ========== UNIVERSAL ROUTING LAYER (Song → Output) ==========

def render_with_song_layer(container: Any, output_format: str = "svg") -> Any:
    """
    MAIN ROUTER: Unified render pipeline with song generation + translation.
    
    Flow:
    1. Detect container type
    2. Generate song internally (CANONICAL representation)
    3. Record operation (track in weight structure)
    4. Translate song to user format
    5. Return formatted output
    
    Args:
        container: Container to render
        output_format: User-expected output format (svg, json, markdown, text, symbol, verse)
    
    Returns:
        Formatted output in requested format
    """
    container_type = getattr(container, '__class__', type(container)).__name__
    
    # Map class names to render types
    type_mapping = {
        "PrimitiveContainer": "primitive",
        "LedgerContainer": "ledger",
        "ImprovedWorldState": "worldstate",
        "ImprovedEntity": "entity",
        "OrientationPrimitives": "orientation",
        "PrimitiveRegistry": "registry"
    }
    
    render_type = type_mapping.get(container_type, "generic")
    
    # LAYER 1: Generate song internally
    song = generate_render_song(render_type, container.__dict__ if hasattr(container, '__dict__') else {})
    
    # LAYER 3: Record operation
    record_render_operation(song, render_type, f"{container_type}_render")
    
    # LAYER 2: Translate to user format
    if output_format in ["json", "markdown", "text", "symbol", "verse"]:
        translation = translate_song_to_format(song, output_format)
        if translation is not None:
            return translation
    
    # Default: Return both canonical and rendered (SVG)
    svg_output = render_container(container)  # Existing render functions
    
    return {
        "canonical": song,  # Internal (for ARIA/recovery)
        "rendered": svg_output,  # User output (SVG)
        "format": output_format
    }


def render_container(container: Any) -> str:
    """
    Render ANY container (PrimitiveContainer, ImprovedWorldState, etc.) universally.
    
    Detects container type and generates appropriate visualization:
    • PrimitiveRegistry → SVG showing spiral hierarchy (all levels)
    • PrimitiveContainer → SVG showing shape, constraints, orientation
    • LedgerContainer → SVG showing causality chains and trust scores
    • ImprovedWorldState → SVG showing spatial layout, cycles, connections
    • OrientationPrimitives → SVG showing anchor vector in 3D field
    • ImprovedEntity → SVG showing position and history
    • Domain-specific → Domain-specific output
    """
    try:
        from UNIVERSAL_ENTITY_CONNECTION_FRAMEWORK import (
            PrimitiveContainer, PrimitiveRegistry, ImprovedWorldState, 
            OrientationPrimitives, ImprovedEntity
        )
    except ImportError:
        return "<error>Could not import containers</error>"
    
    # Special case: PrimitiveRegistry gets spiral hierarchy
    container_type = getattr(container, '__class__', type(container)).__name__
    if container_type == "PrimitiveRegistry" or hasattr(container, 'PRIMITIVES'):
        return render_spiral_hierarchy(container, "Universal Framework Spiral")
    
    # Detect container type by checking for specific attributes
    if isinstance(container, PrimitiveContainer):
        return render_primitive_container(container)
    elif container_type == "LedgerContainer" or hasattr(container, 'ledger_id'):
        return render_ledger_container(container)
    elif isinstance(container, ImprovedWorldState):
        return render_world_state_container(container)
    elif isinstance(container, OrientationPrimitives):
        return render_orientation_container(container)
    elif isinstance(container, ImprovedEntity):
        return render_entity_container(container)
    else:
        return f"<error>Unknown container type: {type(container)}</error>"


def render_primitive_container(container: 'PrimitiveContainer') -> str:
    """Render a PrimitiveContainer showing its shape and constraints."""
    svg_elements = []
    
    # Canvas
    width, height = 800, 600
    svg_elements.append(f'<svg width="{width}" height="{height}" xmlns="http://www.w3.org/2000/svg">')
    svg_elements.append('<style>text { font-family: Arial; font-size: 12px; }</style>')
    
    # Title
    title = escape_xml(f"{container.primitive_name} ({container.primitive_type})")
    svg_elements.append(f'<text x="20" y="30" font-weight="bold" font-size="16">{title}</text>')
    
    # Domain
    svg_elements.append(f'<text x="20" y="55" fill="blue">Domain: {container.domain}</text>')
    
    # Fields list
    y_offset = 80
    svg_elements.append('<text x="20" y="80" font-weight="bold">Fields:</text>')
    for i, (fname, ftype) in enumerate(container.fields.items()):
        required = "●" if fname in container.required_fields else "○"
        svg_elements.append(f'<text x="40" y="{y_offset + (i+1)*20}">{required} {escape_xml(fname)}: {escape_xml(ftype)}</text>')
    
    y_offset += len(container.fields) * 20 + 40
    
    # Constraints
    if container.constraints:
        svg_elements.append(f'<text x="20" y="{y_offset}" font-weight="bold">Constraints:</text>')
        for i, (cname, cval) in enumerate(container.constraints.items()):
            svg_elements.append(f'<text x="40" y="{y_offset + (i+1)*20}">{escape_xml(str(cname))}: {escape_xml(str(cval))}</text>')
    
    # Capabilities
    y_offset += max(1, len(container.constraints)) * 20 + 40
    svg_elements.append(f'<text x="20" y="{y_offset}" font-weight="bold">Capabilities:</text>')
    caps = []
    if container.can_contain_self:
        caps.append("✓ Can self-reference")
    if container.cycle_detection_required:
        caps.append("✓ Cycle detection")
    if container.validated:
        caps.append("✓ Validated")
    if container.locked:
        caps.append("✓ Locked")
    
    for i, cap in enumerate(caps):
        svg_elements.append(f'<text x="40" y="{y_offset + (i+1)*20}">{cap}</text>')
    
    # Orientation anchor (if exists)
    if container.orientation:
        y_offset += len(caps) * 20 + 40
        svg_elements.append(f'<text x="20" y="{y_offset}" font-weight="bold">Anchor:</text>')
        anchor = container.orientation
        svg_elements.append(f'<text x="40" y="{y_offset + 20}">Type: {escape_xml(anchor.anchor_type)}</text>')
        svg_elements.append(f'<text x="40" y="{y_offset + 40}">Vector: ({anchor.anchor_vector[0]:.2f}, {anchor.anchor_vector[1]:.2f}, {anchor.anchor_vector[2]:.2f})</text>')
        svg_elements.append(f'<text x="40" y="{y_offset + 60}">Magnitude: {anchor.magnitude:.2f}</text>')
    
    svg_elements.append('</svg>')
    
    return "\n".join(svg_elements)

@dataclass
class UniversalResult:
    """Universal result wrapper enforcing causality through data types.
    
    Next stage checks if this succeeded before proceeding.
    """
    success: bool
    data: Optional[Dict] = None
    quality_score: float = 0.0
    verification_passed: bool = False
    violations: List[str] = field(default_factory=list)
    stage_name: str = ""
    
    def __post_init__(self):
        # Next stage MUST check this
        if not self.success and not self.violations:
            self.violations.append(f"Stage '{self.stage_name}' failed with no violations recorded")
    
    def failed(self) -> bool:
        return not self.success or self.violations

class Stage1_InputValidator:
    """STAGE 1: VALIDATE - Ensure molecule data is safe"""
    
    @staticmethod
    def validate_molecule(molecule: Molecule) -> UniversalResult:
        """Validate molecule structure before any processing."""
        violations = []
        
        # Check atoms exist
        if not molecule or not molecule.atoms:
            violations.append("No atoms in molecule")
        
        # Check bond references are valid
        for bond_idx, bond in enumerate(molecule.bonds):
            atom1_idx, atom2_idx, bond_order = bond
            if atom1_idx >= len(molecule.atoms) or atom2_idx >= len(molecule.atoms):
                violations.append(f"Bond {bond_idx} references invalid atom indices")
            if bond_order <= 0 or bond_order > 3:
                violations.append(f"Bond {bond_idx} has invalid bond order {bond_order}")
        
        # Check coordinate validity
        for atom_idx, atom in enumerate(molecule.atoms):
            element, x, y, z = atom
            if element not in ["H", "C", "N", "O"]:
                violations.append(f"Atom {atom_idx} has unsupported element {element}")
            if not all(isinstance(v, (int, float)) for v in [x, y, z]):
                violations.append(f"Atom {atom_idx} has non-numeric coordinates")
        
        success = len(violations) == 0
        
        return UniversalResult(
            success=success,
            data={"molecule": molecule, "atom_count": len(molecule.atoms)},
            stage_name="InputValidator"
        ) if success else UniversalResult(
            success=False,
            violations=violations,
            stage_name="InputValidator"
        )


class Stage2_MetricsCalculator:
    """STAGE 2: METRICS - Analyze molecule geometry"""
    
    @staticmethod
    def calculate_metrics(molecule: Molecule) -> UniversalResult:
        """Calculate structural metrics for strategy selection."""
        
        atoms = molecule.atoms
        
        if len(atoms) < 2:
            return UniversalResult(
                success=True,
                data={
                    "spread_factor": 1.0,
                    "density": 1.0,
                    "asymmetry": 0.0,
                    "max_dist": 1.0,
                    "avg_inter_atomic": 1.0,
                    "num_atoms": len(atoms),
                },
                quality_score=1.0,
                stage_name="MetricsCalculator"
            )
        
        # Center of mass
        cx = sum(a[1] for a in atoms) / len(atoms)
        cy = sum(a[2] for a in atoms) / len(atoms)
        cz = sum(a[3] for a in atoms) / len(atoms)
        
        # Max distance from center
        max_dist = max(
            math.sqrt((a[1] - cx)**2 + (a[2] - cy)**2 + (a[3] - cz)**2)
            for a in atoms
        )
        
        # Average inter-atomic distance
        total_dist = 0
        count = 0
        for i in range(len(atoms)):
            for j in range(i + 1, len(atoms)):
                d = math.sqrt(
                    (atoms[i][1] - atoms[j][1])**2 +
                    (atoms[i][2] - atoms[j][2])**2 +
                    (atoms[i][3] - atoms[j][3])**2
                )
                total_dist += d
                count += 1
        avg_inter_atomic = total_dist / count if count > 0 else 1.0
        
        # Density
        field_volume = (max_dist ** 3) * 4 / 3
        density = len(atoms) / max(0.1, field_volume)
        
        # Asymmetry
        distances_from_center = [
            math.sqrt((a[1] - cx)**2 + (a[2] - cy)**2 + (a[3] - cz)**2)
            for a in atoms
        ]
        avg_dist = sum(distances_from_center) / len(distances_from_center)
        variance = sum((d - avg_dist)**2 for d in distances_from_center) / len(distances_from_center)
        asymmetry = math.sqrt(variance) / max(0.1, avg_dist)
        
        # Spread factor
        spread_factor = max_dist / max(0.1, avg_inter_atomic)
        
        metrics = {
            "spread_factor": spread_factor,
            "density": density,
            "asymmetry": asymmetry,
            "max_dist": max_dist,
            "avg_inter_atomic": avg_inter_atomic,
            "num_atoms": len(atoms),
        }
        
        return UniversalResult(
            success=True,
            data=metrics,
            quality_score=1.0,
            stage_name="MetricsCalculator"
        )


class Stage3_StrategySelector:
    """STAGE 3: STRATEGY - Choose rendering approach based on quaternion rotation magnitude"""
    
    @staticmethod
    def select_strategy(metrics_result: UniversalResult) -> UniversalResult:
        """Select rotation strategy based on quaternion, NOT Euler angles (CAUSALITY: requires metrics)."""
        
        if metrics_result.failed():
            return UniversalResult(
                success=False,
                violations=["Metrics stage failed, cannot select strategy"],
                stage_name="StrategySelector"
            )
        
        metrics = metrics_result.data
        quaternion = metrics.get("quaternion")  # Now using quaternion, not Euler angles
        num_atoms = metrics["num_atoms"]
        
        if quaternion is None:
            return UniversalResult(
                success=False,
                violations=["No quaternion in metrics"],
                stage_name="StrategySelector"
            )
        
        # Extract rotation magnitude from quaternion (no gimbal lock!)
        aa = quaternion.to_axis_angle()
        angle_deg = abs(aa['angle_deg'])
        
        # Choose strategy based on quaternion rotation magnitude
        if angle_deg > 180:
            strategy_type = 'quaternion_based_large'
            num_frames = 360
        elif angle_deg > 45:
            strategy_type = 'quaternion_based_medium'
            num_frames = 180
        else:
            strategy_type = 'quaternion_based_small'
            num_frames = 90
        
        # FORMAT SELECTION: Determined by USER INTENT (global decision, not per-molecule complexity)
        # PRINCIPLE: Pick ONE consistent format for entire project, applied universally
        # User intent: GIF → all 9 molecules render as GIF (not mixed formats)
        complexity_score = (num_atoms / 10.0) + (angle_deg / 180.0)
        output_format = "gif"  # USER INTENT: GIF format for all molecules uniformly
        
        strategy = {
            "strategy_type": strategy_type,
            "num_frames": num_frames,
            "quaternion_angle_deg": angle_deg,
            "quaternion_magnitude": float(quaternion.magnitude()),
            "metrics_applied": True,
            "output_format": output_format,
            "complexity_score": complexity_score,
            "gimbal_lock_risk": 0.0,  # Always 0 with quaternions!
        }
        
        return UniversalResult(
            success=True,
            data=strategy,
            quality_score=1.0,
            stage_name="StrategySelector"
        )

@dataclass
class PrimitiveWeights:
    """Unified weighted primitives for all rendering operations."""
    spread_factor: float = 1.0
    density: float = 1.0
    asymmetry: float = 0.0
    num_atoms: int = 1
    max_dist: float = 1.0
    avg_inter_atomic: float = 1.0
    
    @property
    def color_intensity(self) -> float:
        """Weight for color saturation (based on density, using invariance baseline)."""
        return min(1.0, InvarianceConstants.COLOR_INTENSITY_BASELINE + (self.density / 10.0))
    
    @property
    def glow_strength(self) -> float:
        """Weight for glow/blur effects (based on asymmetry)."""
        return min(1.0, 0.3 + self.asymmetry)
    
    @property
    def opacity_base(self) -> float:
        """Base opacity weight (based on spread and density)."""
        return min(1.0, (self.spread_factor / 5.0) * (self.density / 5.0))
    
    @property
    def radius_weight(self) -> float:
        """Weight for radius scaling (based on spread factor)."""
        return min(2.0, 0.8 + (self.spread_factor / 10.0))

        for entity in self.entities:
            if entity.id == entity_id:

        return None

        """Get all connections involving this entity."""
        return [c for c in self.connections 
                if c.entity1_id == entity_id or c.entity2_id == entity_id]

class UniversalRenderer:
    """Universal renderer for any domain (molecules, compute, agents, ledgers, etc).
    
    Uses WorldState containers to render any connected graph with fields.
    Adapts visualization to all available data in the containers for maximum resolution.
    
    Now integrated with HUMAN STANDARDS:
    • Orientations expressed as quaternions (Hamilton convention)
    • Dipoles represented as arrow vectors
    • All rendered containers validated against standards
    """
    
    def __init__(self):
        os.makedirs(OUTPUT_DIR, exist_ok=True)
        self.metrics = {
            "render_time": 0,
            "frames_generated": 0,
            "file_size_kb": 0,
        }
        self.current_weights = PrimitiveWeights()
        self.standards_containers = {}  # Cache for standards-validated containers
    
    def calculate_universal_metrics(self, world: WorldState, quaternion: Quaternion = None) -> UniversalResult:
        """Calculate metrics for ANY domain using entity positions and connections.
        
        Now includes quaternion orientation from HUMAN STANDARDS.
        """
        
        if not world.entities or len(world.entities) < 1:
            return UniversalResult(
                success=True,
                data={
                    "spread_factor": 1.0, "density": 1.0, "asymmetry": 0.0, 
                    "max_dist": 1.0, "avg_inter_distance": 1.0, "num_entities": 0,
                    "quaternion": quaternion or Quaternion(1, 0, 0, 0)  # Identity quaternion
                },
                stage_name="UniversalMetrics"
            )
        
        # Use provided quaternion or create identity
        if quaternion is None:
            quaternion = Quaternion(1, 0, 0, 0)
        
        # Extract positions
        positions = [e.position for e in world.entities]
        
        # Center of mass
        cx = sum(p[0] for p in positions) / len(positions)
        cy = sum(p[1] for p in positions) / len(positions)
        cz = sum(p[2] for p in positions) / len(positions)
        
        # Max distance from center
        max_dist = max(
            math.sqrt((p[0] - cx)**2 + (p[1] - cy)**2 + (p[2] - cz)**2)
            for p in positions
        )
        
        # Average inter-entity distance
        total_dist = 0
        count = 0
        for i in range(len(positions)):
            for j in range(i + 1, len(positions)):
                d = math.sqrt(
                    (positions[i][0] - positions[j][0])**2 +
                    (positions[i][1] - positions[j][1])**2 +
                    (positions[i][2] - positions[j][2])**2
                )
                total_dist += d
                count += 1
        avg_inter_distance = total_dist / count if count > 0 else 1.0
        
        # Density (entities per unit volume)
        field_volume = (max_dist ** 3) * 4 / 3
        density = len(positions) / max(0.1, field_volume)
        
        # Asymmetry (how non-spherical distribution is)
        distances_from_center = [
            math.sqrt((p[0] - cx)**2 + (p[1] - cy)**2 + (p[2] - cz)**2)
            for p in positions
        ]
        avg_dist = sum(distances_from_center) / len(distances_from_center)
        variance = sum((d - avg_dist)**2 for d in distances_from_center) / len(distances_from_center)
        asymmetry = math.sqrt(variance) / max(0.1, avg_dist)
        
        # Spread factor
        spread_factor = max_dist / max(0.1, avg_inter_distance)
        
        # Connection density (how interconnected)
        connection_density = len(world.connections) / max(1, len(world.entities))
        
        metrics = {
            "spread_factor": spread_factor,
            "density": density,
            "asymmetry": asymmetry,
            "max_dist": max_dist,
            "avg_inter_distance": avg_inter_distance,
            "num_entities": len(world.entities),
            "num_connections": len(world.connections),
            "quaternion": quaternion,  # NEW: Standards-based orientation
            "connection_density": connection_density,
        }
        
        return UniversalResult(
            success=True,
            data=metrics,
            quality_score=1.0,
            stage_name="UniversalMetrics"
        )
    
    def _weighted_color(self, base_color: Tuple[int, int, int], intensity_factor: float = 1.0) -> Tuple[int, int, int]:
        """Apply intensity weighting to RGB color."""
        r, g, b = base_color
        factor = min(1.0, self.current_weights.color_intensity * intensity_factor)
        return (int(r * factor), int(g * factor), int(b * factor))
    
    def _weighted_alpha(self, base_alpha: int, opacity_factor: float = 1.0) -> int:
        """Apply opacity weighting to alpha channel."""
        return int(base_alpha * self.current_weights.opacity_base * opacity_factor)
    
    def _weighted_width(self, base_width: float, scale_factor: float = 1.0) -> int:
        """Apply radius weighting to line width."""
        return max(1, int(base_width * self.current_weights.radius_weight * scale_factor))
    
    def _weighted_radius(self, base_radius: float, scale_factor: float = 1.0) -> float:
        """Apply radius weighting to circle/ellipse radius."""
        return base_radius * self.current_weights.radius_weight * scale_factor
    
    def render_world_to_gif(self, world: WorldState, output_filename: str = None, frames: int = 20) -> Tuple[str, UniversalResult]:

            )

                success=False,

            )
    
    def stage5_verify(self, execution_result: UniversalResult) -> UniversalResult:
        """STAGE 5: VERIFY - Quality check (requires Stage 4 execution)."""
        
        if execution_result.failed() or "frames" not in (execution_result.data or {}):
            return UniversalResult(
                success=False,
                violations=["Execution stage failed or no frames available"],
                verification_passed=False,
                stage_name="QualityVerifier"
            )
        
        violations = []
        frames = execution_result.data["frames"]
        
        # Verify frames
        if not frames:
            violations.append("No frames in execution result")
        else:
            for idx, frame in enumerate(frames):
                if not isinstance(frame, Image.Image):
                    violations.append(f"Frame {idx} is not a PIL Image")
                elif frame.size != (400, 400):
                    violations.append(f"Frame {idx} has wrong size {frame.size}")
        
        verification_passed = len(violations) == 0
        
        return UniversalResult(
            success=len(violations) == 0,
            data=execution_result.data,
            quality_score=InvarianceConstants.VERIFICATION_QUALITY_PASS if verification_passed else InvarianceConstants.VERIFICATION_QUALITY_FAIL,
            verification_passed=verification_passed,
            violations=violations,
            stage_name="QualityVerifier"
        )

    def _calculate_rotations(self, strategy: Dict, frame_idx: int, total_frames: int, molecule: Molecule = None) -> Tuple[float, float, float]:
        """Calculate rotation angles: rotate AROUND axis PERPENDICULAR to dipole (the anchor/"pole")
        
        Principle: Dipole/pole stays visible. Molecule rotates around perpendicular axis.
        This keeps the molecular anchor (dipole) stationary while showing 3D structure.
        Camera angle: Isometric view (35.26° elevation, 45° azimuth) for visual clarity
        """
        
        frame_norm = (frame_idx / total_frames) * 2 * math.pi
        
        # Strategy parameters from metrics
        y_rotation_base = strategy["y_rotation_base"]
        x_tilt_frequency = strategy["x_tilt_frequency"]
        x_tilt_amplitude = strategy["x_tilt_amplitude"]
        z_roll_frequency = strategy["z_roll_frequency"]
        z_roll_amplitude = strategy["z_roll_amplitude"]
        
        # ISOMETRIC CAMERA ANGLE (initial orientation for best visual clarity)
        # Isometric: ~35.26° elevation, 45° azimuth
        isometric_elevation = 35.26  # degrees from horizontal
        isometric_azimuth = 45.0      # degrees around vertical
        
        # PRIMARY ROTATION: Around axis PERPENDICULAR to dipole
        # Dipole vector is the anchor/"pole" - should remain visible
        # Find axis perpendicular to it and rotate molecule around that axis
        if molecule:
            # Use parent calculation from universal entity framework
            molecule_atoms = [(elem, x, y, z) for elem, x, y, z in molecule.atoms]
            anchor = calculate_molecular_anchor(molecule_atoms)  # OrientationPrimitives
            dipole_axis = anchor.anchor_vector  # Normalized (dx, dy, dz)
            
            # Find perpendicular axis: cross product with a reference vector
            # If dipole is (dx, dy, dz), perpendicular can be (-dy, dx, 0) when normalized
            # This gives us an axis in the plane perpendicular to dipole
            perp_x = -dipole_axis[1]
            perp_y = dipole_axis[0]
            perp_z = 0.0
            
            # Normalize perpendicular axis
            perp_mag = math.sqrt(perp_x**2 + perp_y**2 + perp_z**2)
            if perp_mag > 0.01:
                perp_x /= perp_mag
                perp_y /= perp_mag
            else:
                # Dipole is pointing along Z axis, use X as perpendicular
                perp_x = 1.0
                perp_y = 0.0
            
            # Rotation amount around perpendicular axis
            perp_rotation_angle = math.degrees(frame_norm) * y_rotation_base
            
            # Map perpendicular rotation to Euler angles for rendering (using invariance scaling)
            # Rotate around perpendicular axis by controlling Y and Z based on perp_x, perp_y
            y_rotation = perp_rotation_angle * perp_x
            z_rotation = isometric_azimuth + (perp_rotation_angle * perp_y * InvarianceConstants.ROTATION_SCALING_FACTOR)
        else:
            # Fallback: standard Y rotation
            y_rotation = math.degrees(frame_norm) * y_rotation_base
            z_rotation = isometric_azimuth
        
        # SECONDARY ROTATIONS: Camera angle + gentle depth variation
        # X tilt: anchored to isometric elevation
        x_rotation = isometric_elevation + (x_tilt_amplitude * math.sin(frame_norm * x_tilt_frequency) * 0.2)
        
        return (y_rotation, x_rotation, z_rotation)
    
    def _calculate_optimal_rotation_params(self, molecule: Molecule) -> Dict[str, float]:
        """
        Calculate field resolution-based rotation parameters.
        Analyzes molecule geometry to determine optimal rotation pattern.
        """
        
        atoms = molecule.atoms
        
        # Calculate geometric properties
        if len(atoms) < 2:
            return {"spread_factor": 1.0, "density": 1.0, "asymmetry": 0.0}
        
        # Center of mass
        cx = sum(a[1] for a in atoms) / len(atoms)
        cy = sum(a[2] for a in atoms) / len(atoms)
        cz = sum(a[3] for a in atoms) / len(atoms)
        
        # Max distance from center (field radius)
        max_dist = max(
            math.sqrt((a[1] - cx)**2 + (a[2] - cy)**2 + (a[3] - cz)**2)
            for a in atoms
        )
        
        # Average inter-atomic distance (field density)
        if len(atoms) > 1:
            total_dist = 0
            count = 0
            for i in range(len(atoms)):
                for j in range(i + 1, len(atoms)):
                    d = math.sqrt(
                        (atoms[i][1] - atoms[j][1])**2 +
                        (atoms[i][2] - atoms[j][2])**2 +
                        (atoms[i][3] - atoms[j][3])**2
                    )
                    total_dist += d
                    count += 1
            avg_inter_atomic = total_dist / count if count > 0 else 1.0
        else:
            avg_inter_atomic = 1.0
        
        # Atom density (atoms per unit volume)
        field_volume = (max_dist ** 3) * 4 / 3  # Approximate sphere volume
        density = len(atoms) / max(0.1, field_volume)
        
        # Asymmetry score (how non-spherical/spread out)
        distances_from_center = [
            math.sqrt((a[1] - cx)**2 + (a[2] - cy)**2 + (a[3] - cz)**2)
            for a in atoms
        ]
        avg_dist = sum(distances_from_center) / len(distances_from_center) if distances_from_center else 1.0
        variance = sum((d - avg_dist)**2 for d in distances_from_center) / len(distances_from_center) if distances_from_center else 0
        asymmetry = math.sqrt(variance) / max(0.1, avg_dist)
        
        # Spread factor: how much to spread rotation based on geometry
        spread_factor = max_dist / max(0.1, avg_inter_atomic)
        
        return {
            "spread_factor": spread_factor,
            "density": density,
            "asymmetry": asymmetry,
            "max_dist": max_dist,
            "avg_inter_atomic": avg_inter_atomic,
            "num_atoms": len(atoms),
        }
    
    def _calculate_field_resolution_rotation(self, molecule: Molecule, frame_idx: int, total_frames: int) -> Tuple[float, float, float]:
        """
        Calculate rotation angles based on field resolution and geometry.
        Creates deterministic, geometry-aware rotation pattern.
        """
        
        params = self._calculate_optimal_rotation_params(molecule)
        
        spread = params["spread_factor"]
        density = params["density"]
        asymmetry = params["asymmetry"]
        num_atoms = params["num_atoms"]
        
        # Normalize frame to 0-2π
        frame_norm = (frame_idx / total_frames) * 2 * math.pi
        
        # FIELD RESOLUTION ROTATION PATTERN:
        # High spread → faster Y rotation (spread molecules need more tumbling)
        # High density → more X tilt (packed molecules need vertical separation)
        # High asymmetry → more Z roll (asymmetric molecules benefit from rolling)
        
        # Y rotation: primary, scaled by field spread
        y_rotation = math.degrees(frame_norm) * (0.8 + spread * 0.3)
        
        # X tilt: frequency scaled by density, amplitude by asymmetry (using invariance constants)
        x_tilt_frequency = InvarianceConstants.FREQUENCY_BASE + density * InvarianceConstants.FREQUENCY_DENSITY_SCALE
        x_tilt_amplitude = InvarianceConstants.TILT_AMPLITUDE_BASE + (asymmetry * InvarianceConstants.TILT_ASYMMETRY_SCALE)
        x_rotation = x_tilt_amplitude * math.sin(frame_norm * x_tilt_frequency)
        
        # Z roll: frequency scaled by num_atoms, phase different from X
        z_roll_frequency = 1.5 + (num_atoms / 10)
        z_roll_amplitude = 5 + (asymmetry * 15)
        z_rotation = z_roll_amplitude * math.cos(frame_norm * z_roll_frequency)
        
        return (y_rotation, x_rotation, z_rotation)

    def _create_frame_unified_field(self, molecule: Molecule, rotation_angle_y: float, 
                                     rotation_angle_x: float, rotation_angle_z: float,
                                     frame_idx: int, total_frames: int,
                                     anchor: OrientationPrimitives, field_config: Dict,
                                     weights: PrimitiveWeights = None) -> Image.Image:
        """Create frame with UNIFIED FIELD approach - FIELD AND ATOMS ROTATE TOGETHER:
        
        Both the field and atoms are part of ONE system that rotates as a whole.
        NOT: field static + atoms rotating separately.
        
        1. BACKGROUND: Atoms + bonds show internal structural detail
        2. FOREGROUND: Field blob shows overall charged entity
        Both rotate together around dipole axis.
        """
        
        if weights is None:
            weights = self.current_weights
        
        # Canvas size
        width, height = 400, 400
        img = Image.new('RGB', (width, height), color=(255, 255, 255))
        
        # Center
        cx, cy = width // 2, height // 2
        scale = 80
        
        # ===== CRITICAL: Rotate dipole vector with frame rotations =====
        # The field orientation changes with rotation angles
        rotated_dipole = self._rotate_dipole_vector(
            field_config["dipole_vector"], 
            rotation_angle_y, rotation_angle_x, rotation_angle_z
        )
        
        # ===== BACKGROUND: Structural Detail (atoms, bonds) =====
        # Rotate and project atoms using SAME rotation angles
        rotated_atoms = self._rotate_atoms(molecule.atoms, rotation_angle_y, rotation_angle_x, rotation_angle_z)
        projected = self._project_atoms(rotated_atoms, scale, cx, cy)
        
        # Draw structural detail layers FIRST (background)
        # LAYER 1: Draw bonds (structural framework)
        draw = ImageDraw.Draw(img)
        self._draw_bonds_with_gradient_glow(img, draw, projected, molecule, weights)
        
        # LAYER 2: Draw atoms as interior detail
        self._draw_atoms_with_multi_layer_glow(img, draw, projected, molecule, weights)
        
        # LAYER 3: Draw electron density clouds (field merging)
        self._draw_electron_density_clouds(img, projected, molecule, weights)
        
        # ===== FOREGROUND: Molecule's Unified Field (PRIMARY OBJECT) =====
        # The entire molecule is ONE field entity - drawn ON TOP of structure
        # Field orientation follows rotated dipole axis
        # Updated field_config with rotated dipole
        rotated_field_config = field_config.copy()
        rotated_field_config["dipole_vector"] = rotated_dipole
        self._draw_unified_field_foreground(img, molecule, anchor, rotated_field_config, cx, cy, scale, weights)
        
        # FINAL LAYER: Draw info text overlay
        self._draw_info_with_glow(img, draw, img.size[0], img.size[1], rotation_angle_y, molecule, frame_idx, total_frames, weights)
        
        return img
    
    def _rotate_dipole_vector(self, dipole: Tuple[float, float, float], 
                              angle_y: float, angle_x: float, angle_z: float) -> Tuple[float, float, float]:
        """Rotate dipole vector using same rotation angles as atoms.
        
        This ensures field orientation follows the same rotation as internal structure.
        Returns rotated dipole vector (already normalized by magnitude preservation).
        """
        # Convert angles to radians
        rad_x = math.radians(angle_x)
        rad_y = math.radians(angle_y)
        rad_z = math.radians(angle_z)
        
        x, y, z = dipole
        
        # Rotate around X axis
        y_x = y * math.cos(rad_x) - z * math.sin(rad_x)
        z_x = y * math.sin(rad_x) + z * math.cos(rad_x)
        
        # Rotate around Y axis
        x_xy = x * math.cos(rad_y) + z_x * math.sin(rad_y)
        z_xy = -x * math.sin(rad_y) + z_x * math.cos(rad_y)
        
        # Rotate around Z axis
        x_final = x_xy * math.cos(rad_z) - y_x * math.sin(rad_z)
        y_final = x_xy * math.sin(rad_z) + y_x * math.cos(rad_z)
        
        return (x_final, y_final, z_xy)
    
    def _draw_unified_field_foreground(self, img: Image.Image, molecule: Molecule, 
                                       anchor: OrientationPrimitives, field_config: Dict,
                                       cx: int, cy: int, scale: float,
                                       weights: PrimitiveWeights):
        """Draw molecule as unified field FOREGROUND object.
        
        The molecule is one cohesive field entity:
        - Colored by partial charges (red = negative, blue = positive)
        - Shaped by molecular geometry (dipole orientation)
        - Strength by molecular anchor magnitude
        
        This is the PRIMARY visualization - what we're looking at.
        """
        draw = ImageDraw.Draw(img, 'RGBA')
        
        # Extract field properties
        dipole_vector = field_config["dipole_vector"]
        dipole_magnitude = field_config["dipole_magnitude"]
        partial_charges = field_config["partial_charges"]
        h_bonding_sites = field_config["h_bonding_sites"]
        
        # Normalize dipole vector for drawing
        if dipole_magnitude > 1e-6:
            dipole_norm = tuple(v / dipole_magnitude for v in dipole_vector)
        else:
            dipole_norm = (0, 0, 1)
        
        # FIELD SHAPE: Ellipsoid elongated along dipole axis
        # Major axis (along dipole): scaled by dipole magnitude + weights
        major_axis = scale * 1.5 * (1.0 + dipole_magnitude * 0.3) * weights.radius_weight
        # Minor axes: scaled by weights
        minor_axis = scale * (1.0 + weights.density * 0.2)
        
        # FIELD COLOR: Blend colors from partial charges
        # Negative (O, N) → red, Positive (H) → blue, Neutral (C) → gray
        avg_charge = 0.0
        charge_count = 0
        for charge_key, charge_val in partial_charges.items():
            avg_charge += charge_val
            charge_count += 1
        avg_charge = avg_charge / max(1, charge_count)
        
        # Color based on polarity
        if avg_charge < -0.1:
            # Negative field - red
            field_r, field_g, field_b = 255, 100, 100
        elif avg_charge > 0.1:
            # Positive field - blue
            field_r, field_g, field_b = 100, 100, 255
        else:
            # Neutral field - gray
            field_r, field_g, field_b = 180, 180, 180
        
        # FIELD INTENSITY: Based on molecular weight and density
        field_intensity = int(180 * weights.color_intensity * (0.7 + weights.density * 0.3))
        field_color = (field_r, field_g, field_b, field_intensity)
        
        # Draw main field blob as ellipse
        # Oriented along dipole vector in projection
        angle = math.atan2(dipole_norm[1], dipole_norm[0])
        angle_deg = math.degrees(angle)
        
        # Draw field as large ellipse (main shape)
        bbox_left = cx - major_axis

    def _rotate_atoms(self, atoms: List[Tuple[str, float, float, float]], 
                      angle_y: float, angle_x: float = 0, angle_z: float = 0) -> List[Tuple[str, float, float, float]]:
        """Rotate atoms around X, Y, Z axes (Euler angles for 3D tumble)."""
        
        angle_rad_x = math.radians(angle_x)
        angle_rad_y = math.radians(angle_y)
        angle_rad_z = math.radians(angle_z)
        
        rotated = []
        
        for element, x, y, z in atoms:
            # Rotate around X axis
            y_x = y * math.cos(angle_rad_x) - z * math.sin(angle_rad_x)
            z_x = y * math.sin(angle_rad_x) + z * math.cos(angle_rad_x)
            
            # Rotate around Y axis
            x_xy = x * math.cos(angle_rad_y) + z_x * math.sin(angle_rad_y)
            z_xy = -x * math.sin(angle_rad_y) + z_x * math.cos(angle_rad_y)
            
            # Rotate around Z axis
            x_xyz = x_xy * math.cos(angle_rad_z) - y_x * math.sin(angle_rad_z)
            y_xyz = x_xy * math.sin(angle_rad_z) + y_x * math.cos(angle_rad_z)
            
            rotated.append((element, x_xyz, y_xyz, z_xy))
        
        return rotated
    
    def _project_atoms(self, atoms: List[Tuple[str, float, float, float]], 
                      scale: float, cx: float, cy: float) -> List[Tuple[float, float, float]]:
        """Project 3D atoms to 2D screen coordinates."""
        
        projected = []
        
        for element, x, y, z in atoms:
            # Perspective projection (simple)
            depth = 2.0 + z
            if depth < 0.1:
                depth = 0.1
            
            screen_x = cx + (x * scale) / depth
            screen_y = cy + (y * scale) / depth
            
            projected.append((screen_x, screen_y, depth))
        
        return projected

    def _draw_dipole_vectors(self, img: Image.Image, draw: ImageDraw.ImageDraw, 
                            projected: List, molecule: Molecule, weights: PrimitiveWeights):
        """Draw dipole moment visualization with weighted primitives."""
        
        # Assign charges for dipole calculation
        charges = {
            "O": -1.0, "N": -0.5, "C": 0.2, "H": 0.1
        }
        
        # Calculate molecular dipole
        total_x, total_y = 0, 0
        total_charge = 0
        
        for atom_idx, atom_proj in enumerate(projected):
            x, y, depth = atom_proj
            element = molecule.atoms[atom_idx][0]
            charge = charges.get(element, 0.0)
            
            total_x += x * charge
            total_y += y * charge
            total_charge += abs(charge)
        
        if total_charge > 0:
            dipole_x = total_x / total_charge
            dipole_y = total_y / total_charge
            
            # Find molecular center
            center_x = sum(p[0] for p in projected) / len(projected)
            center_y = sum(p[1] for p in projected) / len(projected)
            
            # Draw dipole vector - weighted by glow_strength for asymmetry prominence
            arrow_length = 40 * weights.glow_strength  # Scale by asymmetry
            arrow_x = center_x + dipole_x * 0.2
            arrow_y = center_y + dipole_y * 0.2
            
            if (dipole_x**2 + dipole_y**2) > 1:
                # Dipole line color weighted for intensity
                dipole_color = self._weighted_color((255, 0, 0), weights.glow_strength)
                dipole_width = self._weighted_width(4, weights.glow_strength)  # Thicker when asymmetric
                
                # Draw dipole vector with glow layers
                for glow_layer in range(2):
                    glow_color = self._weighted_color((255, 100, 100), (1 - glow_layer / 2) * weights.glow_strength)
                    glow_width = dipole_width + (glow_layer * 2)
                    draw.line(
                        [(center_x, center_y), (arrow_x, arrow_y)],
                        fill=glow_color,
                        width=glow_width
                    )
                
                # Draw main dipole line
                draw.line(
                    [(center_x, center_y), (arrow_x, arrow_y)],
                    fill=dipole_color,
                    width=dipole_width
                )
                
                # Draw arrowhead - weighted for visibility
                angle = math.atan2(arrow_y - center_y, arrow_x - center_x)
                arrowhead_size = 8 * weights.glow_strength
                for da in [-0.3, 0.3]:
                    ux = arrow_x - arrowhead_size * math.cos(angle + da)
                    uy = arrow_y - arrowhead_size * math.sin(angle + da)
                    draw.line([(arrow_x, arrow_y), (ux, uy)], fill=dipole_color, width=self._weighted_width(2, weights.glow_strength))
    
    def _draw_valence_electron_dots(self, img: Image.Image, draw: ImageDraw.ImageDraw, 
                                    projected: List, molecule: Molecule, weights: PrimitiveWeights):
        """Draw Lewis-style valence electron dots with weighted primitives."""
        
        valence = {
            "H": 1, "C": 4, "N": 5, "O": 6
        }
        
        for atom_idx, atom_proj in enumerate(projected):
            x, y, depth = atom_proj
            element = molecule.atoms[atom_idx][0]
            num_valence = valence.get(element, 0)
            
            # Draw electron dots around atom (Lewis structure style)
            dot_radius = self.ATOM_SIZES.get(element, 10) + 6
            dots_per_side = (num_valence + 3) // 4
            
            for dot_idx in range(num_valence):
                angle = (dot_idx / max(1, num_valence)) * 2 * math.pi
                dot_x = x + dot_radius * math.cos(angle)
                dot_y = y + dot_radius * math.sin(angle)
                
                draw.ellipse(
                    [(dot_x - 2, dot_y - 2), (dot_x + 2, dot_y + 2)],
                    fill=(255, 100, 100)
                )
    
    def _draw_resonance_structures(self, img: Image.Image, draw: ImageDraw.ImageDraw, 
                                   projected: List, molecule: Molecule, weights: PrimitiveWeights):
        """Indicate resonance structures with arrows and weighted primitives."""
        
        if "Benzene" in molecule.name or len(projected) >= 4:
            # Find molecular bounds
            xs = [p[0] for p in projected]
            ys = [p[1] for p in projected]
            
            if xs and ys:
                min_x, max_x = min(xs), max(xs)
                min_y, max_y = min(ys), max(ys)
                
                # Draw resonance indicator arrows
                for i in range(2):
                    start_x = min_x - 25 + (i * 50)
                    start_y = max_y + 15
                    
                    # Arrow shaft
                    draw.line(
                        [(start_x, start_y), (start_x + 20, start_y)],
                        fill=(150, 100, 200),
                        width=2
                    )
                    
                    # Arrowhead
                    draw.polygon(
                        [(start_x + 20, start_y), (start_x + 16, start_y - 3), (start_x + 16, start_y + 3)],
                        fill=(150, 100, 200)
                    )
