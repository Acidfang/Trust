```python
def get_server_health():
    """Poll /api/health endpoint - returns dict or None"""
    try:
        response = requests.get("http://127.0.0.1:8081/api/health", timeout=2)
        return response.json() if response.ok else {"status": "error"}
    except:
        return {"status": "unreachable"}
```
