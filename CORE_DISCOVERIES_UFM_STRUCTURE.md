def query_any_domain(root_node, path_expression):
    """
    Works on ANYTHING because everything is a binary tree
    """
    current = root_node
    for element in path_expression:
        # Binary decision at each node: follow left (0) or right (1)
        if element == 0:
            current = current.left
        else:
            current = current.right
    return current
